\begin{tikzpicture}[font=\small]
% --- 定义式 ---
\node[anchor=west] at (0,3.5) {
  $\mathrm{EED}(i,j,L)\;=\;d_{ij}\times\dfrac{\alpha+\beta L}{\alpha}

% --- 基准：几何距离 ---
\draw[dashed, gray] (6,0.15) -- (6,2.15);
\node[gray,anchor=south] at (6,0.05) {几何距离 $d_{ij}$};

% --- 满载：等效电量距离更长 ---
\draw[very thick, {Stealth[length=3mm]}-{Stealth[length=3mm]}] (0,1.75) -- (12,1.75);
\node[anchor=west,align=left] at (12.2,1.75)
  {满载出发：$d_{ij}\bigl(1+\tfrac{\beta}{\alpha}L_{\max}\bigr)$};

% --- 空载：退化为几何距离 ---
\draw[very thick, {Stealth[length=3mm]}-{Stealth[length=3mm]}] (0,0.65) -- (6,0.65);
\node[anchor=west] at (6.2,0.65) {空载出发：$=d_{ij}$};
\end{tikzpicture}
